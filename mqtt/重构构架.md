## 🎯 架构重构方案（职责分离）

当前 `Client` 结构体是**上帝对象**，违反单一职责原则。需要拆分为三个独立组件。

---

## 问题分析

### 当前结构体（违反单一职责）

```go
type Client struct {
    client        mqtt.Client          // MQTT 通信
    config        *config.Config
    dataQueue     *queue.Queue         // 重试队列
    monitor       *monitor.Monitor     // 监控指标
    analyzer      *analyzer.Analyzer   // 数据分析
    filterManager *filter.FilterManager // 数据过滤
    broadcaster   broadcast.Broadcaster // WebSocket 广播
    writePool     *pool.Pool           // 并发写入

    mu             sync.Mutex          // 批量缓冲
    pendingRecords []*map[string]any
    lastBatchTime  time.Time
    batchTimer     *time.Timer

    registeredDevices map[string]bool  // 设备管理
    muDevices         sync.Mutex
}
```

**问题**：一个结构体承担了 MQTT 通信、数据解析、过滤、分析、批量写入、广播、监控、设备管理等 8 种职责。

---

## 重构方案

### 职责拆分

| 组件 | 职责 | 依赖 |
|------|------|------|
| **MQTTClient** | 连接、订阅、发布、消息通道 | paho-mqtt |
| **BatchWriter** | 批量缓冲、Pool 并发写入、重试 | queue, pool, database |
| **MessageProcessor** | 解析、过滤、分析、广播 | analyzer, filter, broadcaster, monitor |

### 架构图

```
┌──────────────────────────────────────────────────────┐
│                  MessageProcessor                    │
│  职责：解析 → 过滤 → 分析 → 广播                      │
│  依赖：analyzer, filter, broadcaster, monitor         │
└──────────────┬───────────────────────────────────────┘
               │ 提交 records
               ▼
┌──────────────────────────────────────────────────────┐
│                   BatchWriter                        │
│  职责：批量缓冲 → Pool 并发 → 写入 → 失败重试          │
│  依赖：pool, queue, database                          │
└──────────────┬───────────────────────────────────────┘
               │ 消费消息
               ▲
┌──────────────┴───────────────────────────────────────┐
│                   MQTTClient                         │
│  职责：连接 → 订阅 → 接收 → 重连恢复                   │
│  依赖：paho-mqtt                                      │
└──────────────────────────────────────────────────────┘
```

## 1. MQTTClient（纯通信层）

```go
package mqtt

type MQTTClient struct {
    client     mqtt.Client
    brokerURL  string
    clientID   string
    username   string
    password   string
    onMessage  mqtt.MessageHandler  // 消息回调（由外部注入）
    subscribed []string             // 记录订阅主题
    muSub      sync.Mutex
    monitor    *monitor.Monitor
}

func NewMQTTClient(cfg *config.Config, onMessage mqtt.MessageHandler, monitor *monitor.Monitor) (*MQTTClient, error) {
    c := &MQTTClient{
        brokerURL: cfg.MQTTBroker,
        clientID:  cfg.ClientID,
        username:  cfg.MQTTUsername,
        password:  cfg.MQTTPassword,
        onMessage: onMessage,
        monitor:   monitor,
    }

    opts := mqtt.NewClientOptions()
    opts.AddBroker(c.brokerURL)
    opts.SetClientID(c.clientID)
    opts.SetCleanSession(true)
    opts.SetAutoReconnect(true)
    opts.SetMaxReconnectInterval(time.Minute * 5)

    if c.username != "" {
        opts.SetUsername(c.username)
        opts.SetPassword(c.password)
    }

    // 遗嘱消息
    opts.SetWill("edgex/events/status", buildOfflinePayload(c.clientID), 0, false)

    // 连接回调
    opts.SetOnConnectHandler(c.onConnect)
    opts.SetConnectionLostHandler(c.onConnectionLost)

    c.client = mqtt.NewClient(opts)
    token := c.client.Connect()
    token.Wait()
    if token.Error() != nil {
        return nil, fmt.Errorf("failed to connect: %v", token.Error())
    }

    return c, nil
}

func (c *MQTTClient) onConnect(client mqtt.Client) {
    log.Println("MQTT broker connected")
    if c.monitor != nil {
        c.monitor.SetMQTTConnectionStatus(true)
    }
    c.publishOnlineStatus()
    c.restoreSubscriptions(client)
}

func (c *MQTTClient) onConnectionLost(client mqtt.Client, err error) {
    log.Printf("MQTT connection lost: %v", err)
    if c.monitor != nil {
        c.monitor.SetMQTTConnectionStatus(false)
    }
}

func (c *MQTTClient) Subscribe(topics []string) error {
    c.muSub.Lock()
    defer c.muSub.Unlock()

    for _, topic := range topics {
        token := c.client.Subscribe(topic, 1, c.onMessage)
        token.Wait()
        if token.Error() != nil {
            return fmt.Errorf("subscribe %s failed: %v", topic, token.Error())
        }
    }
    c.subscribed = append(c.subscribed, topics...)
    log.Printf("Subscribed to topics: %v", topics)
    return nil
}

func (c *MQTTClient) restoreSubscriptions(client mqtt.Client) {
    c.muSub.Lock()
    topics := make([]string, len(c.subscribed))
    copy(topics, c.subscribed)
    c.muSub.Unlock()

    for _, topic := range topics {
        client.Subscribe(topic, 1, c.onMessage)
    }
    log.Println("Subscriptions restored")
}

func (c *MQTTClient) Publish(topic string, qos byte, payload any) error {
    token := c.client.Publish(topic, qos, false, payload)
    token.Wait()
    return token.Error()
}

func (c *MQTTClient) Disconnect() {
    c.client.Disconnect(250)
}
```

## 2. BatchWriter（批量写入层）

```go
package mqtt

type BatchWriter struct {
    writePool     *pool.Pool
    dataQueue     *queue.Queue
    monitor       *monitor.Monitor
    broadcaster   broadcast.Broadcaster
    analyzer      *analyzer.Analyzer

    mu             sync.Mutex
    pendingRecords []*map[string]any
    lastBatchTime  time.Time
    batchTimer     *time.Timer
}

func NewBatchWriter(dataQueue *queue.Queue, monitor *monitor.Monitor,
    broadcaster broadcast.Broadcaster, analyzer *analyzer.Analyzer) *BatchWriter {
    
    return &BatchWriter{
        writePool:      pool.NewPoolForIO(),
        dataQueue:      dataQueue,
        monitor:        monitor,
        broadcaster:    broadcaster,
        analyzer:       analyzer,
        pendingRecords: make([]*map[string]any, 0, batchSize),
        lastBatchTime:  time.Now(),
    }
}

func (w *BatchWriter) Enqueue(records []*map[string]any) {
    w.mu.Lock()
    defer w.mu.Unlock()

    w.pendingRecords = append(w.pendingRecords, records...)

    shouldFlush := len(w.pendingRecords) >= batchSize ||
        time.Since(w.lastBatchTime) >= batchTime*time.Millisecond

    if shouldFlush {
        w.flush()
        return
    }

    if w.batchTimer == nil {
        w.batchTimer = time.AfterFunc(batchTime*time.Millisecond, func() {
            w.mu.Lock()
            defer w.mu.Unlock()
            if len(w.pendingRecords) > 0 {
                w.flush()
            } else {
                w.batchTimer = nil
            }
        })
    }
}

func (w *BatchWriter) flush() {
    records := w.pendingRecords
    w.pendingRecords = make([]*map[string]any, 0, batchSize)
    w.lastBatchTime = time.Now()
    if w.batchTimer != nil {
        w.batchTimer.Stop()
        w.batchTimer = nil
    }

    w.writePool.Submit(func() {
        w.doWrite(records)
    })
}

func (w *BatchWriter) doWrite(records []*map[string]any) {
    if len(records) == 0 {
        return
    }

    if w.monitor != nil {
        w.monitor.IncrementDatabaseOperations()
        w.monitor.IncrementTotalRecordsStored(int64(len(records)))
    }

    err := database.BatchInsertWithRetry(database.Table, records, 3, 2*time.Second)
    if err != nil {
        w.handleWriteError(err, records)
        return
    }

    if w.monitor != nil {
        w.monitor.IncrementMQTTMessagesProcessed()
    }

    // 广播
    if w.broadcaster != nil {
        w.broadcastData("device_data", map[string]any{
            "deviceName": (*records[0])["deviceName"],
            "records":    records,
        })
    }

    // 分析
    if w.analyzer != nil && w.analyzer.IsEnabled() && len(records) <= 50 {
        w.analyzeData(records, "")
    }
}

func (w *BatchWriter) handleWriteError(err error, records []*map[string]any) {
    if w.monitor != nil {
        w.monitor.RecordError("storage_error", err.Error())
    }
    if w.dataQueue != nil {
        w.dataQueue.Enqueue(records)
    }
}

func (w *BatchWriter) Stop() {
    w.mu.Lock()
    if len(w.pendingRecords) > 0 {
        w.flush()
    }
    w.mu.Unlock()
    w.writePool.Stop()
}
```

## 3. MessageProcessor（消息处理层）

```go
package mqtt

type MessageProcessor struct {
    monitor       *monitor.Monitor
    filterManager *filter.FilterManager
}

func NewMessageProcessor(monitor *monitor.Monitor) *MessageProcessor {
    return &MessageProcessor{
        monitor:       monitor,
        filterManager: filter.NewFilterManager(),
    }
}

func (p *MessageProcessor) Handler(enqueue func([]*map[string]any)) mqtt.MessageHandler {
    return func(client mqtt.Client, msg mqtt.Message) {
        p.recordMessageReceived(len(msg.Payload()))

        event, err := edgex.ProcessMessage(msg.Payload())
        if err != nil {
            log.Printf("Failed to process message: %v", err)
            return
        }
        if event == nil {
            p.monitor.IncrementMQTTMessagesFiltered()
            return
        }

        records := p.processReadings(event)
        if len(records) > 0 {
            enqueue(records)
        }
    }
}

func (p *MessageProcessor) recordMessageReceived(payloadSize int) {
    if p.monitor != nil {
        p.monitor.IncrementMQTTMessagesReceived()
        p.monitor.IncrementDataReceivedBytes(int64(payloadSize))
    }
}

func (p *MessageProcessor) processReadings(event *edgex.EdgeXEvent) []*map[string]any {
    records := make([]*map[string]any, 0, len(event.Readings))

    for _, reading := range event.Readings {
        value := common.ParseValue(reading.Value)

        if isInvalidValue(value) {
            if p.monitor != nil {
                p.monitor.RecordError("invalid_value_discarded", 
                    fmt.Sprintf("Device: %s, Reading: %s", event.DeviceName, reading.ResourceName))
                p.monitor.IncrementMQTTMessagesFiltered()
            }
            continue
        }

        if p.monitor != nil {
            floatValue := toFloat64(value)
            p.monitor.UpdateDeviceStatus(event.DeviceName, reading.ResourceName, floatValue)
        }

        if !p.shouldStore(event.DeviceName, reading.ResourceName, value) {
            continue
        }

        records = append(records, &map[string]any{
            "id":         reading.ID,
            "deviceName": event.DeviceName,
            "reading":    reading.ResourceName,
            "value":      value,
            "valueType":  reading.ValueType,
            "baseType":   reading.BaseType,
            "timestamp":  reading.Origin,
            "metadata":   string(reading.Metadata),
        })
    }

    return records
}

func (p *MessageProcessor) shouldStore(deviceName, resourceName string, value any) bool {
    if p.filterManager != nil {
        return p.filterManager.ShouldStore(deviceName, resourceName, value)
    }
    return true
}
```

## 4. 组装（main.go）

```go
func main() {
    cfg := config.Load()

    // 1. 创建批量写入器
    batchWriter := mqtt.NewBatchWriter(
        queue.New(),
        monitor.New(),
        broadcast.New(),
        analyzer.New(cfg),
    )

    // 2. 创建消息处理器（注入 enqueue 回调）
    processor := mqtt.NewMessageProcessor(batchWriter.monitor)
    handler := processor.Handler(batchWriter.Enqueue)

    // 3. 创建 MQTT 客户端（注入 handler）
    mqttClient, err := mqtt.NewMQTTClient(cfg, handler, batchWriter.monitor)
    if err != nil {
        log.Fatal(err)
    }

    // 4. 订阅主题
    mqttClient.Subscribe(getTopics(cfg))

    // 5. 优雅停机
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
    <-sigChan

    log.Println("Shutting down...")
    batchWriter.Stop()
    mqttClient.Disconnect()
    log.Println("Shutdown complete")
}
```

## 对比

| 维度 | 重构前 | 重构后 |
|------|--------|--------|
| 职责数 | 8 种 | 每种 1-2 种 |
| 字段数 | 14 个 | 每个 5-8 个 |
| 可测试性 | 难（需 mock 8 个依赖） | 易（独立测试） |
| 可维护性 | 低（改一处影响全局） | 高（边界清晰） |
| 复用性 | 无 | 各组件可独立复用 |

---

## 结论

重构后的架构：

1. **MQTTClient** - 只管通信，不关心业务
2. **BatchWriter** - 只管写入，不关心来源
3. **MessageProcessor** - 只管处理，不关心存储

通过**回调函数**和**接口**解耦，符合单一职责原则，且保持简单。

## 🏗️ 生产级接口抽象架构

本架构采用**接口驱动 + 事件路由 + 优雅停机**的设计，彻底解决之前版本的所有问题，并满足生产环境要求。

---

## 核心接口定义

```go
package mqtt

import (
	"context"
	"sync/atomic"
)

type Message struct {
	Topic     string
	Payload   []byte
	Timestamp int64
}

type MessageHandler func(ctx context.Context, msg *Message)

type Publisher interface {
	Publish(topic string, qos byte, retained bool, payload interface{}) error
}

type Subscriber interface {
	Subscribe(topics map[string]byte) error
	Unsubscribe(topics ...string) error
	Messages() <-chan *Message
}

type Client interface {
	Publisher
	Subscriber
	Connect() error
	Disconnect(ctx context.Context) error
	Metrics() *Metrics
}

type Metrics struct {
	Received  atomic.Int64
	Processed atomic.Int64
	Failed    atomic.Int64
	Dropped   atomic.Int64
}

func (m *Metrics) Snapshot() map[string]int64 {
	return map[string]int64{
		"received":  m.Received.Load(),
		"processed": m.Processed.Load(),
		"failed":    m.Failed.Load(),
		"dropped":   m.Dropped.Load(),
	}
}
```

## 1. MQTT 通信层（接口实现）

```go
package mqtt

import (
	"context"
	"fmt"
	"log"
	"sync"
	"time"

	mqttlib "github.com/eclipse/paho.mqtt.golang"
)

type Config struct {
	BrokerURL      string
	ClientID       string
	Username       string
	Password       string
	WillTopic      string
	WillMessage    string
	KeepAlive      int
	CleanSession   bool
	MaxChannelSize int
	ReconnectDelay time.Duration
}

type clientImpl struct {
	config         *Config
	mqttClient     mqttlib.Client
	msgChan        chan *Message
	ctx            context.Context
	cancel         context.CancelFunc
	subscribed     map[string]byte
	mu             sync.RWMutex
	metrics        *Metrics
	done           chan struct{}
}

func NewClient(cfg *Config) Client {
	if cfg.MaxChannelSize <= 0 {
		cfg.MaxChannelSize = 1000
	}
	if cfg.ReconnectDelay <= 0 {
		cfg.ReconnectDelay = 5 * time.Second
	}
	ctx, cancel := context.WithCancel(context.Background())
	return &clientImpl{
		config:     cfg,
		msgChan:    make(chan *Message, cfg.MaxChannelSize),
		ctx:        ctx,
		cancel:     cancel,
		subscribed: make(map[string]byte),
		metrics:    &Metrics{},
		done:       make(chan struct{}),
	}
}

func (c *clientImpl) Connect() error {
	opts := mqttlib.NewClientOptions()
	opts.AddBroker(c.config.BrokerURL)
	opts.SetClientID(c.config.ClientID)
	if c.config.Username != "" {
		opts.SetUsername(c.config.Username)
		opts.SetPassword(c.config.Password)
	}
	opts.SetCleanSession(c.config.CleanSession)
	opts.SetKeepAlive(time.Duration(c.config.KeepAlive) * time.Second)
	opts.SetWill(c.config.WillTopic, c.config.WillMessage, 0, false)
	opts.SetAutoReconnect(true)
	opts.SetMaxReconnectInterval(c.config.ReconnectDelay)
	opts.SetOnConnectHandler(c.onConnect)
	opts.SetConnectionLostHandler(c.onConnectionLost)

	c.mqttClient = mqttlib.NewClient(opts)
	token := c.mqttClient.Connect()
	if !token.WaitTimeout(10 * time.Second) {
		return fmt.Errorf("timeout connecting to MQTT broker")
	}
	if token.Error() != nil {
		return fmt.Errorf("failed to connect: %w", token.Error())
	}

	log.Println("[MQTT] connected successfully")
	return nil
}

func (c *clientImpl) Subscribe(topics map[string]byte) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	for topic, qos := range topics {
		token := c.mqttClient.Subscribe(topic, qos, c.messageHandler)
		token.Wait()
		if token.Error() != nil {
			return fmt.Errorf("subscribe to %s failed: %w", topic, token.Error())
		}
		c.subscribed[topic] = qos
	}
	log.Printf("[MQTT] subscribed: %v", topics)
	return nil
}

func (c *clientImpl) Unsubscribe(topics ...string) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	token := c.mqttClient.Unsubscribe(topics...)
	token.Wait()
	if token.Error() != nil {
		return fmt.Errorf("unsubscribe failed: %w", token.Error())
	}
	for _, t := range topics {
		delete(c.subscribed, t)
	}
	log.Printf("[MQTT] unsubscribed: %v", topics)
	return nil
}

func (c *clientImpl) Publish(topic string, qos byte, retained bool, payload interface{}) error {
	token := c.mqttClient.Publish(topic, qos, retained, payload)
	token.Wait()
	return token.Error()
}

func (c *clientImpl) Messages() <-chan *Message {
	return c.msgChan
}

func (c *clientImpl) Disconnect(ctx context.Context) error {
	c.cancel()
	if c.mqttClient.IsConnected() {
		c.mqttClient.Disconnect(250)
	}
	close(c.done)
	close(c.msgChan)
	log.Println("[MQTT] disconnected")
	return nil
}

func (c *clientImpl) Metrics() *Metrics {
	return c.metrics
}

func (c *clientImpl) onConnect(_ mqttlib.Client) {
	log.Println("[MQTT] reconnected, restoring subscriptions...")

	c.mu.RLock()
	topics := make(map[string]byte, len(c.subscribed))
	for k, v := range c.subscribed {
		topics[k] = v
	}
	c.mu.RUnlock()

	if len(topics) > 0 {
		if err := c.resubscribe(topics); err != nil {
			log.Printf("[MQTT] restore subscriptions failed: %v", err)
		} else {
			log.Println("[MQTT] subscriptions restored")
		}
	}
}

func (c *clientImpl) resubscribe(topics map[string]byte) error {
	for topic, qos := range topics {
		token := c.mqttClient.Subscribe(topic, qos, c.messageHandler)
		token.Wait()
		if token.Error() != nil {
			return token.Error()
		}
	}
	return nil
}

func (c *clientImpl) onConnectionLost(_ mqttlib.Client, err error) {
	log.Printf("[MQTT] connection lost: %v", err)
}

func (c *clientImpl) messageHandler(_ mqttlib.Client, msg mqttlib.Message) {
	c.metrics.Received.Add(1)

	internalMsg := &Message{
		Topic:     msg.Topic(),
		Payload:   msg.Payload(),
		Timestamp: time.Now().UnixMilli(),
	}

	select {
	case c.msgChan <- internalMsg:
	case <-c.ctx.Done():
		return
	default:
		c.metrics.Dropped.Add(1)
		log.Printf("[MQTT] channel full, dropping message from %s", msg.Topic())
	}
}
```

## 2. 事件路由器（基于 Pool 的事件分发）

```go
package router

import (
	"context"
	"log"
	"regexp"
	"sync"

	"sfsEdgeStore/mqtt"
	"sfsEdgeStore/pool"
)

type routeEntry struct {
	pattern *regexp.Regexp
	handler mqtt.MessageHandler
}

type EventRouter struct {
	mqttClient mqtt.Client
	routes     []routeEntry
	mu         sync.RWMutex
	pool       *pool.Pool
	metrics    *mqtt.Metrics
	ctx        context.Context
}

func NewEventRouter(client mqtt.Client, poolSize int) *EventRouter {
	return &EventRouter{
		mqttClient: client,
		pool:       pool.NewPoolWithSize(poolSize),
		metrics:    client.Metrics(),
	}
}

func NewEventRouterWithPool(client mqtt.Client, p *pool.Pool) *EventRouter {
	return &EventRouter{
		mqttClient: client,
		pool:       p,
		metrics:    client.Metrics(),
	}
}

func (r *EventRouter) AddRoute(pattern string, handler mqtt.MessageHandler) error {
	re, err := regexp.Compile(pattern)
	if err != nil {
		return err
	}

	r.mu.Lock()
	defer r.mu.Unlock()

	r.routes = append(r.routes, routeEntry{pattern: re, handler: handler})
	return nil
}

func (r *EventRouter) Start(ctx context.Context) {
	r.ctx = ctx
	go r.dispatcher(ctx)
}

func (r *EventRouter) dispatcher(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			log.Println("[Router] dispatcher stopped")
			return
		case msg, ok := <-r.mqttClient.Messages():
			if !ok {
				log.Println("[Router] message channel closed")
				return
			}
			r.submit(msg)
		}
	}
}

func (r *EventRouter) submit(msg *mqtt.Message) {
	r.pool.Submit(func() {
		r.dispatch(msg)
	})
}

func (r *EventRouter) dispatch(msg *mqtt.Message) {
	r.mu.RLock()
	routes := r.routes
	r.mu.RUnlock()

	for _, route := range routes {
		if route.pattern.MatchString(msg.Topic) {
			r.metrics.Processed.Add(1)

			func() {
				defer func() {
					if err := recover(); err != nil {
						r.metrics.Failed.Add(1)
						log.Printf("[Router] handler panic: %v", err)
					}
				}()
				route.handler(r.ctx, msg)
			}()
			return
		}
	}

	log.Printf("[Router] no route for topic: %s", msg.Topic)
}

func (r *EventRouter) Stop() {
	r.pool.Stop()
	log.Println("[Router] pool stopped")
}
```

## 2.1 Pool 实现（项目复用）

```go
package pool

import (
	"runtime"
	"sync"
)

type Pool struct {
	wg   sync.WaitGroup
	work chan func()
	size int
}

func NewPool() *Pool {
	return NewPoolWithSize(runtime.NumCPU())
}

func NewPoolWithSize(size int) *Pool {
	if size <= 0 {
		size = runtime.NumCPU()
	}
	p := &Pool{
		work: make(chan func(), size),
		size: size,
	}
	for i := 0; i < size; i++ {
		p.wg.Add(1)
		go p.worker()
	}
	return p
}

func NewPoolForCPU() *Pool {
	return NewPoolWithSize(runtime.NumCPU())
}

func NewPoolForIO() *Pool {
	return NewPoolWithSize(runtime.NumCPU() * 4)
}

func NewPoolForNetwork() *Pool {
	return NewPoolWithSize(runtime.NumCPU() * 10)
}

func (p *Pool) worker() {
	defer p.wg.Done()
	for task := range p.work {
		task()
	}
}

func (p *Pool) Submit(task func()) {
	p.work <- task
}

func (p *Pool) Stop() {
	close(p.work)
	p.wg.Wait()
}

func (p *Pool) Size() int {
	return p.size
}
```

## 3. 业务处理器（接口实现）

```go
package processor

import (
	"context"
	"log"

	"your-project/mqtt"
)

type Analyzer interface {
	Process(topic string, payload []byte) (interface{}, error)
}

type Filter interface {
	ShouldFilter(topic string, payload []byte) bool
}

type Broadcaster interface {
	Broadcast(result interface{}) error
}

type MessageProcessor struct {
	analyzer    Analyzer
	filter      Filter
	broadcaster Broadcaster
}

func NewMessageProcessor(analyzer Analyzer, filter Filter, broadcaster Broadcaster) *MessageProcessor {
	return &MessageProcessor{
		analyzer:    analyzer,
		filter:      filter,
		broadcaster: broadcaster,
	}
}

func (p *MessageProcessor) Handler() mqtt.MessageHandler {
	return func(ctx context.Context, msg *mqtt.Message) {
		if p.filter.ShouldFilter(msg.Topic, msg.Payload) {
			return
		}

		result, err := p.analyzer.Process(msg.Topic, msg.Payload)
		if err != nil {
			log.Printf("[Processor] analyze failed: %v", err)
			return
		}

		if err := p.broadcaster.Broadcast(result); err != nil {
			log.Printf("[Processor] broadcast failed: %v", err)
		}
	}
}
```

## 4. 主程序（优雅停机）

```go
package main

import (
	"context"
	"log"
	"os"
	"os/signal"
	"syscall"
	"time"

	"your-project/mqtt"
	"your-project/processor"
	"your-project/router"
)

func main() {
	client := mqtt.NewClient(&mqtt.Config{
		BrokerURL:    "tcp://localhost:1883",
		ClientID:     "sfs-edge-store",
		Username:     "user",
		Password:     "pass",
		KeepAlive:    60,
		CleanSession: true,
	})

	if err := client.Connect(); err != nil {
		log.Fatalf("[Main] connect failed: %v", err)
	}

	topics := map[string]byte{
		"sensor/data":   1,
		"device/status": 0,
	}
	if err := client.Subscribe(topics); err != nil {
		log.Fatalf("[Main] subscribe failed: %v", err)
	}

	proc := processor.NewMessageProcessor(
		newAnalyzer(),
		newFilter(),
		newBroadcaster(),
	)

	eventRouter := router.NewEventRouter(client, 4)

	if err := eventRouter.AddRoute(`^sensor/.*$`, proc.Handler()); err != nil {
		log.Fatalf("[Main] add route failed: %v", err)
	}

	ctx, cancel := context.WithCancel(context.Background())
	eventRouter.Start(ctx)

	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)

	<-sigChan
	log.Println("[Main] shutdown signal received")

	cancel()
	eventRouter.Stop()

	shutdownCtx, shutdownCancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer shutdownCancel()

	if err := client.Disconnect(shutdownCtx); err != nil {
		log.Printf("[Main] disconnect error: %v", err)
	}

	log.Printf("[Main] metrics: %v", client.Metrics().Snapshot())
	log.Println("[Main] graceful shutdown completed")
}

func newAnalyzer() processor.Analyzer    { return &processor.AnalyzerImpl{} }
func newFilter() processor.Filter        { return &processor.FilterImpl{} }
func newBroadcaster() processor.Broadcaster { return &processor.BroadcasterImpl{} }
```

## 架构优势总结

| 维度 | 改进前 | 改进后 |
|------|--------|--------|
| 依赖耦合 | 具体类型 `*mqtt.Client` | 接口 `Publisher`/`Subscriber`/`Client` |
| CleanSession | 与手动重订阅矛盾 | 统一 `true` + 客户端负责重订阅 |
| 优雅停机 | `time.Sleep` 不可靠 | `context` + `WaitGroup` + `done channel` |
| 错误处理 | `log.Fatal` 直接退出 | 返回 `error`，由调用方决定 |
| 消息路由 | 单通道单消费者 | 正则路由 + 多 worker 并发 |
| Metrics | TODO 未实现 | `atomic.Int64` 完整统计 |
| 防 Panic | 无 | `recover` 捕获 handler panic |
| 测试能力 | 难以 mock | 接口驱动，轻松 mock |
| 并发管理 | Event Router 自建 worker | 复用项目 Pool，职责分离 |
| 背压控制 | channel 满则丢弃 | pool.work channel 天然阻塞背压 |
